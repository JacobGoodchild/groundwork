Variational Auto-Encoder

Auto-Regressive DGMs

\mathbf{x} = (x_1, x_2, \ldots, x_d)
p(\mathbf{x}) = \prod_{i=1}^{d} p(x_i \mid x_1, x_2, \ldots, x_{i-1}) = \prod_{i=1}^{d} p(x_i \mid \mathbf{x}_{<i})
\mathbf{x}_{<i} = (x_1, x_2, \ldots, x_{i-1})^\top

Diffusion process

q(x_0,\ldots,x_N) = q(x_0)\,q(x_1 \mid x_0)\cdots q(x_N \mid x_{N-1})
q(x_n \mid x_{n-1}) = \N(\sqrt{\alpha_n}\,x_{n-1}, \beta_n I), \qquad \alpha_n = 1-\beta_n
x_N \approx \N(0, I)

Diffusion SDEs and ODEs

\mathrm{d}\bm{x}_t = \left[f(t)\bm{x}_t + \frac{g^2(t)}{\sigma_t}\bm{\epsilon}_\theta(\bm{x}_t,t)\right]\mathrm{d}t + g(t)\,\mathrm{d}\bar{\bm{w}}_t, \quad \bm{x}_T \sim \N(\bm{0}, \tilde\sigma^2\bm{I})
\frac{\mathrm{d}\bm{x}_t}{\mathrm{d}t} = f(t)\bm{x}_t + \frac{g^2(t)}{2\sigma_t}\bm{\epsilon}_\theta(\bm{x}_t,t), \quad \bm{x}_T \sim \N(\bm{0}, \tilde\sigma^2\bm{I})

Training DPMs

q(x_0,\ldots,x_N) \approx p(x_0,\ldots,x_N)
p(x_{n-1} \mid x_n) = \N(\mu_n(x_n), \sigma_n^2 I)
\min_{\mu_n(\cdot),\,\sigma_n^2} KL(q(x_{0:N}) \,\|\, p(x_{0:N})) \;\Leftrightarrow\; \max_{\mu_n(\cdot),\,\sigma_n^2} \E_q \log \frac{p(x_{0:N})}{q(x_{1:N} \mid x_0)}
\min_{s_n(\cdot)} \E_n \bar\beta_n \E_{q_n(x_n)} \|s_n(x_n) - \nabla \log q_n(x_n)\|^2
\sigma_n^2 = \beta_n
\mu_n(x_n) = \frac{1}{\sqrt{\alpha_n}}\left(x_n + \beta_n s_n(x_n)\right)

Analytic-DPM

p(x_{n-1} \mid x_n) = \N(\mu_n(x_n), \Sigma_n(x_n)), \qquad \Sigma_n(x_n) = \sigma_n^2
\hat\epsilon_n(x_n) \approx \mathbf{E}_{q(x_0 \mid x_n)}[\epsilon_n]
\mu_n^*(x_n) = \frac{1}{\sqrt{\alpha_n}}\left(x_n - \frac{\beta_n}{\sqrt{\bar\beta_n}}\mathbf{E}_{q(x_0 \mid x_n)}[\epsilon_n]\right)
\sigma_n^{*2} = \frac{\beta_n}{\alpha_n}\left(1 - \frac{\beta_n}{\bar\beta_n}\E_{q_n(x_n)}\frac{\left\|\mathbf{E}_{q(x_0 \mid x_n)}[\epsilon_n]\right\|^2}{d}\right)

Log-SNR

q_{0t}(\bm{x}_t \mid \bm{x}_0) = \N(\bm{x}_t \mid \alpha(t)\bm{x}_0, \sigma^2(t)\bm{I})
\text{SNR} = \alpha_t^2/\sigma_t^2, \qquad \lambda_t := \log(\alpha_t/\sigma_t)
f(t) = \frac{\mathrm{d}\log\alpha_t}{\mathrm{d}t}, \qquad g^2(t) = -2\sigma_t^2\frac{\mathrm{d}\lambda_t}{\mathrm{d}t}
\bm{x}_t = e^{\int_s^t f(\tau)\mathrm{d}\tau}\bm{x}_s + \int_s^t \left(e^{\int_\tau^t f(r)\mathrm{d}r}\frac{g^2(\tau)}{2\sigma_\tau}\bm{\epsilon}_\theta(\bm{x}_\tau,\tau)\right)\mathrm{d}\tau
\bm{x}_t = \frac{\alpha_t}{\alpha_s}\bm{x}_s - \alpha_t \int_{\lambda_s}^{\lambda_t} e^{-\lambda}\hat{\bm{\epsilon}}_\theta(\hat{\bm{x}}_\lambda,\lambda)\,\mathrm{d}\lambda

DPM-Solver

\bm{x}_{t_{i-1}\to t_i} = \frac{\alpha_{t_i}}{\alpha_{t_{i-1}}}\tilde{\bm{x}}_{t_{i-1}} - \alpha_{t_i}\int_{\lambda_{t_{i-1}}}^{\lambda_{t_i}} e^{-\lambda}\hat{\bm{\epsilon}}_\theta(\hat{\bm{x}}_\lambda,\lambda)\,\mathrm{d}\lambda
\hat{\bm{\epsilon}}_\theta(\hat{\bm{x}}_\lambda,\lambda) = \sum_{n=0}^{k-1}\frac{(\lambda-\lambda_{t_{i-1}})^n}{n!}\hat{\bm{\epsilon}}_\theta^{(n)}(\hat{\bm{x}}_{\lambda_{t_{i-1}}},\lambda_{t_{i-1}}) + \mathcal{O}((\lambda-\lambda_{t_{i-1}})^k)
\bm{x}_{t_{i-1}\to t_i} = \frac{\alpha_{t_i}}{\alpha_{t_{i-1}}}\tilde{\bm{x}}_{t_{i-1}} - \alpha_{t_i}\sum_{n=0}^{k-1} \hat{\bm{\epsilon}}_\theta^{(n)}(\hat{\bm{x}}_{\lambda_{t_{i-1}}},\lambda_{t_{i-1}}) \int_{\lambda_{t_{i-1}}}^{\lambda_{t_i}} e^{-\lambda}\frac{(\lambda-\lambda_{t_{i-1}})^n}{n!}\,\mathrm{d}\lambda + \mathcal{O}(h_i^{k+1})
